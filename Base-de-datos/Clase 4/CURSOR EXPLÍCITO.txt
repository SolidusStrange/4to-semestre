Ejercicio
- Generar modelo que permita almacenar datos de alumno y asignatura asociada al alumno. 
- Aplicar consulta JOIN y usar cursor para mostrar los resultados.


-- 1. Crear tablas
DROP TABLE ESTUDIANTE;

CREATE TABLE ESTUDIANTE(
    rut VARCHAR2(12) PRIMARY KEY,
    primer_nombre VARCHAR2(50) NOT NULL,
    cod_asignatura NUMBER NOT NULL
);    

CREATE TABLE ASIGNATURA(
    cod_asignatura NUMBER NOT NULL,
    nombre_asignatura VARCHAR2(50) NOT NULL
);

-- 3. GENERAMOS BLOQUE ANONIMO

DECLARE
    cursor c_estudiante_asignatura is
        SELECT
            e.rut,
            e.primer_nombre,
            e.cod_asignatura,
            nvl(a.nombre_asignatura, 'SIN ASIGNATURA') AS nombre_asignatura
        FROM ESTUDIANTE e
        LEFT JOIN ASIGNATURA a 
        ON e.cod_asignatura = a.cod_asignatura;
BEGIN
    
    FOR registro in c_estudiante_asignatura
        LOOP
            dbms_output.put_line('Informacion estudiante asignatura: ' || registro.rut);
            dbms_output.put_line('Informacion estudiante asignatura: ' || registro.primer_nombre);
            dbms_output.put_line('Informacion estudiante asignatura: ' || registro.cod_asignatura);
            dbms_output.put_line('Informacion estudiante asignatura: ' || registro.nombre_asignatura);
        END LOOP;

END;
